\begin{frame}[t]{Local specification checks}

$$
\begin{aligned}
\muhatmrp = \meantar \yhat_j =
    \underbrace{
        \expect{\post}{\meantar \m(\theta^\trans \x_j)}.
    }_{\textrm{
        \shortstack{Want to check specification \\ without re--running MCMC}
    }}
\end{aligned}
$$


Some existing local approaches:
%
\begin{itemize}
\item Specify an alternative model, maybe test with Bayes factor
\parencite{dinoia:2025:LikelihoodDistortionBayesian2025,cabral:2025:RobustnessModelChecking2025}
%
\begin{itemize}
    \item Bayes factors can depend strongly on priors (Lindley's paradox)
    \item Articulating a family of models is inconvenient
    \item Doesn't take the target into account
\end{itemize}
%
\item Use out-of-sample checking \parencite{vehtariPracticalBayesianModel2017,kuhUsingLeaveoneoutCross2024}
\begin{itemize}
    \item The task is not to predict on a new IID survey datapoint
\end{itemize}
\end{itemize}

\pause
The model is definitely misspecified, we want to check whether it's \emph{importantly} misspecified.

\end{frame}

\begin{frame}[t]{Local specification checks}

$$
\begin{aligned}
\muhatmrp = \meantar \yhat_j =
    \underbrace{
        \expect{\post}{\meantar \m(\theta^\trans \tarcol{\x_j})}.
    }_{\textrm{
        \shortstack{Want to check specification \\ without re--running MCMC}
    }}
\end{aligned}
$$

Why do we care about correct specification? We want to use 
$\psur(\x)$ to learn about $\ptar(\x)$.\\[1em]

$$
\begin{aligned}
    \textrm{Correct specification} && \Rightarrow &&
    \only<1>{
        {\textrm{Regression estimates are invariant to }\psur(\x)}
    }
    \only<2->{
        \underbrace
        {\textrm{Regression estimates are invariant to }\psur(\x)}_{
                    \textrm{This is what we actually want --- let's check this directly.}
        }
    }
\end{aligned}
$$



\onslide<3->{
Let $\p(\x,\y)$ denote a generic joint density of $\x, \y$.\\
Let $\psurhat$ the empirical distribution on the survey data.
%
\begin{enumerate}
\item Write our estimand as a functional $\mu(\p)$ that takes in a density and returns a MrP estimate
\item Define a covariate density perturbation $\w(\x)$
\item Define $\p_\epsilon(\x, \y) = 
    \p(\x, \y) + \epsilon \w(\x) \p(\y | \x) = 
    \left(\p(\x) + \epsilon \w(\x) \right) \p(\y | \x)$
\item Compute the derivative $\Delta(\w, \p) = \partial \mu(\p_\epsilon) / \partial \epsilon$
\item Plug the empirical distribution into $\Delta(\w, \psurhat)$ and check whether $\approx 0$
\end{enumerate}
}
%


\onslide<4->{
Lots of similar
ideas in causal inference,
kernel learning,
transfer learning 
\parencite{shimodaira:2000:ImprovingPredictiveInference2000,
  buhlmann:2020:InvarianceCausalityRobustness2020,
  rojas-carulla::InvariantModelsCausal,
  muandet:2020:KernelConditionalMoment2020}.  This is just a local Bayesian diagnostics version.
}

\end{frame}

\def\sensitivitysetup{
Let the regression log likelihood be $\ell(\y \vert \x, \theta)$, and
fix $\w(\x)$.  Then define
$$
\post \propto \exp\left(\sumsur \ell(\y_i \vert \x_i, \theta) \right) \prior(\theta)
\textrm{ and }
\posteps \propto 
\post \exp\left(\sumsur \epsilon \w(\x_i) \ell(\y_i \vert \x_i, \theta) \right)
$$
}

\def\sensitivitytheorem{
\theorem{
    \textbf{Theorem:} 
    \parencite{diaconis:1986:ConsistencyBayesEstimates1986,gustafson:1996:localposterior,giordano:2018:covariances}:\\
Consider a family of distributions
$
\peps(\theta) \propto \p(\theta) \exp(\epsilon \h(\theta))
$
and a function $\phi(\theta)$.  By interchange of integration and differentiation,
$$
\fracat{\partial\expect{\peps(\theta)}{\phi(\theta)}}
       {\partial \epsilon}{\epsilon = 0} =
\cov{\p(\theta)}{\phi(\theta), \h(\theta)}.
$$
%
}
}

\begin{frame}[t]{Bayesian sensitivity analysis}

\sensitivitysetup{}

\pause 

\sensitivitytheorem{}

\pause 

Our diagnostic is then given by
$$
\begin{aligned}
    \Delta(\data) :=
    \fracat{\partial \muhatmrp}{\partial \epsilon}{\epsilon = 0} =
    \cov{\post}{
        \underbrace{
            \meantar \m(\theta^\trans \x_j)
        }_{\textrm{Estimand}}, 
        \underbrace{
            \sumsur \w(\x_i) \ell(\y_i \vert \x_i, \theta)
        }_{\textrm{Log model perturbation}}
        }
\end{aligned}
$$
We expect $\Delta(\w, \data)$ to be \emph{close} to zero when the model is correctly
specified.

\end{frame}

\begin{frame}[t]{Bootstrapping Bayesian sensitivity analysis}

We expect $\Delta(\w, \data)$ to be \emph{close} to zero when the model is correctly
specified.\\[1em]

But on any actual dataset, $\Delta(\w, \data)$ will typically not be precisely
zero, even under correct specification.

\only<2>{
\begin{center}
\begin{minipage}{0.6\textwidth}
\OttoGraphZero{}
\end{minipage}
\end{center}

Simulation data: Logistic regression with five active regressors, each
with covariate shift.
$$
\begin{aligned}
    \textrm{Model 0} \textrm{ (correct)}:{}&  \texttt{\y} \sim \textrm{Logistic}(\texttt{1 + X1 + X2 + X3 + X4 + X5})\\
    \textrm{Model 1} \textrm{ (misspecified)}:{}&  \texttt{\y} \sim \textrm{Logistic}(\texttt{1 + X3 + X4 + X5})\\
\end{aligned}
$$
}

\only<3>{
%
To assess ``closeness to zero,'' compare with $\Delta(\data^*)$, where $\data^*$ 
is a bootstrap sample.\\[1em]
%
\begin{itemize}
\item Compute $\Delta(\data)$ on the original dataset
\item For $b=1, \ldots, 100$:
%
\begin{enumerate}
\item Sample $\data^*_b := (\x_1^*, \y_1^*), \ldots (\x_\nsur^*, \y_\nsur^*)$
     (parametric or nonparametric bootstrap)
\item Run MCMC to estimate $\p(\theta | \data^*_b)$ \textbf{(time consuming!)}
\item Use the draws to estimate $\Delta(\data^*_b)$
\end{enumerate}
%
\item Compare $\Delta$ with distribution of $\Delta(\data^*_b)$.
%
\end{itemize}
%
}

\def\generalizedpost{
Define the ``generalized posterior'' functional \parencite{giordano:2023:bayesij}:
%
\begin{align*}
    %
    \T(\p, N) := \frac{
        \int \phi(\theta) \exp\left( 
            N \int \ell(\yn \vert \x, \theta) \p(\xn, \yn) d\xn \, d\yn
        \right)
        \prior(\theta) d\theta
    }
    {
    \int \exp\left( 
            N \int \ell(\yn \vert \x, \theta) \p(\xn, \yn) d\xn \, d\yn
    \right)
    \prior(\theta) d\theta
    }.
    %
\end{align*}
%
}
