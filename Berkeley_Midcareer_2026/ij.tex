\begin{frame}{Why frequentist variance?}

For the remainder of the talk, focus on \emph{fast approximations} to the
\emph{frequentist} sampling distribution under
$\x_i,\y_i \iid \psur(\x, \y)$ of Bayesian posterior quantities like
our diagnostic.\\[1em]

We will form \emph{local approximations} computable only from 
\emph{the original posterior}.\\[1em]

\pause
\textbf{No need to commit to our covariate shift diagnostic to care!}

%
\begin{itemize}
\item Frequentist variablity of \emph{any} model checking diagnostic is interesting
\item Frequentist variablity meaningful under misspecification
  \parencite{huggins:2023:bayesbag,wang:2026:AccurateLargesampleUncertainty2026,
         luo:2026:RobustStandardErrors2026,ji:2025:ValidStandardErrors2025}
\end{itemize}
%


%We will compare with computationally expensive full bootstrap.
\end{frame}

\begin{frame}{Data re-weighting (again)}

\sensitivitysetup{}

% \sensitivitytheorem{}

\pause

If we let $\w(\x_i)$ denote a \emph{bootstrap weight}, then
$\peps$ at $\epsilon = 1$ is the bootstrapped posterior!

\pause

We can form a series approximation:
$$
\begin{aligned}
\expect{\p(\theta \vert \data^*)}{\phi(\theta)} \approx{}&
\expect{\p(\theta \vert \data)}{\phi(\theta)} +
\fracat{\partial \expect{\peps(\theta)}{\phi(\theta)}}{\partial \epsilon}{\epsilon=0}(1 - 0)
\onslide<4->{
\\={}&
\expect{\p(\theta \vert \data)}{\phi(\theta)} +
\sumsur
\underbrace{
    \fracat{\partial \expect{\peps(\theta)}{\phi(\theta)}}{\partial \w_i}{\w_i=0}
}_{\textrm{"Empirical influence function" }\frac{\infl_i}{\nsur}}
 \w_i
}
\onslide<5->{
\\={}&
\expect{\p(\theta \vert \data)}{\phi(\theta)} +
\meansur \infl_i \w_i
}
\end{aligned}
$$

\vspace{1em}
\onslide<6->{
    If this series is accurate, then we can compute $\infl_i$ using the
    original posterior and quickly approximate bootstrap draws.\\[1em]

    \textbf{Task: }Prove that the series is asymptoticallly accurate enough for hypothesis testing.
}

\end{frame}

\begin{frame}[t]{Bayesian von--Mises Expansion}

\generalizedpost{}

Note that $\expect{\post}{\phi(\theta)} = \T(\psurhat, \nsur)$
%
% For $\p$ fixed, as $N \rightarrow \infty$, $\p(\theta \vert \p, N) \rightarrow $a point
% (Bayesian concentration).\\
% As $N \rightarrow \infty$, $\psurhat \rightarrow \psur$ (frequentist concentration).\\
% \pause
%

\pause

To derive frequentist variance estimates, we study the \textit{von Mises expansion}
\parencite{mises:1947:asymptotic}.\\
Let $\tilde\p^\t  = \t \psurhat + (1 - \t) \psur$.   Then 
form the series expansion in $\t$:
%
\begin{align*}
    \sqrt{\nsur} \left(\expect{\post}{\phi(\theta)} - \T(\psur, \nsur) \right) ={}&
    \sqrt{\nsur} \left(\T(\psurhat, \nsur) - \T(\psur, \nsur) \right) \\={}&
        {\sqrt{\nsur} \fracat{\partial \T(\tilde\p^\t, \nsur)}{\partial \t}{\t = 0}}(1 - 0)
         + {\err(\ttil)} 
         & \textrm{(}\ttil \in [0,1]\textrm{)}
\\={}&
\undernote{
    {\sqrt{\nsur} \sumn (\infl_i - \overline{\infl})} }
    {\textrm{{Infinitesimal jackknife estimator}}} + o_p(1) 
    + {\err(\ttil)}.
\end{align*}
%
% where
% $$
% \infl_i := \nsur \cov{\post}{\phi(\theta), \ell(\y_i \vert \x_i)}
% \textrm{ and }.
% $$
%
\end{frame}

\begin{frame}[t]{Bayesian von--Mises Expansion Results}
%
\begin{align*}
    \sqrt{\nsur} \left(\expect{\post}{\phi(\theta)} - \T(\psur, \nsur) \right) 
={}&
    {\sqrt{\nsur} \sumn (\infl_i - \overline{\infl})} + o_p(1) 
     + {\err(\ttil)}.
\end{align*}
%
\theorem{
    \textbf{Theorem: }\parencite{giordano:2023:bayesij} (sketch):\\
    Under Bernstein-von Mises-like conditions,
    % \textbf{(Consistency of the von--Mises expansion in finite dimensions) }
    %
% \begin{align*}
    $\sup_{\ttil \in [0,1]} | {\err(\ttil)} | \overset{p}{\rightarrow} 0$.
    % \quad\textrm{in the Bayesian von--Mises expansion}.
% \end{align*}
%
}

\theorem{
    \textbf{Corollary: } \parencite{mises:1947:asymptotic} Let 
         $\meansur (\infl_i - \overline{\infl})^2 \overset{p}{\rightarrow} V$.
By the CLT applied to $\infl_i$,
\begin{align*}
     \sqrt{\nsur} \left(\expect{\post}{\phi(\theta)} - \T(\psur, \nsur) \right)
     \overset{d}{\rightarrow}
     \mathcal{N}\left(
        0, V
     \right).
\end{align*}
}

\pause

To compute and study the parametric bootstrap instead is a simple change:
%
$$
\posteps \propto 
\post \exp\left(\sumsur \epsilon 
    \left( \ell(\y^*_i \vert \x_i, \theta) - 
           \ell(\y_i \vert \x_i, \theta)\right) \right).
$$
% 
(See, e.g., \textcite{giordano:2026:mrplew} Theorem 4.1).

\pause
\textbf{Note:} The von-Mises expansion is stronger than necessary for 
consistency \parencite{giordano:2023:bayesij}.  But it can be used to
prove \emph{lower bounds}
on $\err(\ttil)$ and so \emph{inconsistency} when no BVM
theorem holds.

\end{frame}
