\begin{frame}[t]{Why frequentist variance?}

For the remainder of the talk, focus on \emph{fast approximations} to the
\emph{frequentist} sampling distribution under
$\x_i,\y_i \iid \psur(\x, \y)$ of Bayesian posterior quantities like
our diagnostic.\\[1em]

We will form \emph{local approximations} computable only from 
\emph{the original posterior}.\\[1em]

\pause
\textbf{No need to commit to our covariate shift diagnostic to care!}

%
\begin{itemize}
\item Frequentist variablity of \emph{any} model checking diagnostic is interesting
\item Frequentist variablity remains meaningful under model misspecification
  \citep{huggins:2023:bayesbag,wang:2026:AccurateLargesampleUncertainty2026,
         luo:2026:RobustStandardErrors2026,ji:2025:ValidStandardErrors2025}
\end{itemize}
%

\pause

We will compare with computationally expensive bootstrap:
%
\begin{itemize}
\item Compute $\Delta$ on the original dataset
\item For $B=1, \ldots, 100$:
%
\begin{enumerate}
\item Sample $(\x_1^*, \y_1^*), \ldots (\x_\nsur^*, \y_\nsur^*)$
     (parametric or nonparametric bootstrap)
\item Run MCMC to estimate $\p(\theta | (\x_1^*, \y_1^*), \ldots (\x_\nsur^*, \y_\nsur^*))$
\item Use the draws to estimate $\Delta^*_B$
\end{enumerate}
%
\item Compare $\Delta$ with distribution of $\Delta^*_B$.
%
\end{itemize}
%
\end{frame}

\begin{frame}[t]{Bayesian von--Mises Expansion}

\generalizedpost{}

Note that $\expect{\post}{\phi(\theta)} = \T(\psurhat, \nsur)$
%
% For $\p$ fixed, as $N \rightarrow \infty$, $\p(\theta \vert \p, N) \rightarrow $a point
% (Bayesian concentration).\\
% As $N \rightarrow \infty$, $\psurhat \rightarrow \psur$ (frequentist concentration).\\
% \pause
%

\pause

To derive frequentist variance estimates, we study the \textit{von Mises expansion}
\citep{mises:1947:asymptotic}.\\
Let $\p_\t  = \t \psurhat + (1 - \t) \psur$.   Then form the series expansion in $\t$:
%
\begin{align*}
    \sqrt{\nsur} \left(\expect{\post}{\g(\theta)} - \T(\psur, \nsur) \right) ={}&
    \sqrt{\nsur} \left(\T(\psurhat, \nsur) - \T(\psur, \nsur) \right) \\={}&
        \blue{\sqrt{\nsur} \fracat{\partial \T(\p_\t, \nsur)}{\partial \t}{\t = 0}}(1 - 0)
         + \red{\err(\ttil)}
\\={}&
\undernote{
    \blue{\sqrt{\nsur} \sumn (\infl_i - \overline{\infl})} }
    {\textrm{\blue{Infinitesimal jackknife estimator}}} + o_p(1) 
    + \red{\err(\ttil)}.
\end{align*}
%
where
$$
\infl_i := \nsur \cov{\post}{\g(\theta), \ell(\y_i \vert \x_i)}
\textrm{ and }\ttil \in [0,1].
$$
%
\end{frame}

\begin{frame}[t]{Bayesian von--Mises Expansion Results}
%
\begin{align*}
    \sqrt{\nsur} \left(\expect{\post}{\g(\theta)} - \T(\psur, \nsur) \right) 
={}&
    \blue{\sqrt{\nsur} \sumn (\infl_i - \overline{\infl})} + o_p(1) 
    & + \red{\err(\ttil)}.
\end{align*}
%


% \begin{itemize}
%     \item Assume (sketch): A well--behaved MAP \textit{maximum a posteriori} estimator $\thetahat$ exists.
%     % \begin{itemize}
%     %     \item The dimension of $\theta$ is fixed as $N \rightarrow \infty$
%     %     \item The expected log likelihood has a strict maximum at $\thetatrue$
%     %     \item The observed log likelihood statisfies $\thetahat \rightarrow \thetatrue$
%     %     \item The expected log likelihood Hessian is negative definite at $\thetatrue$ 
%     % \end{itemize}
%     \item Assume (sketch): We can apply standard Berstein-von Mises asymptotics.
%     % \begin{itemize}
%     % \item The log prior and log likelihood are four times continuously differentiable
%     % \item The prior is proper, and a technical set of prior expectations are finite
%     % \item The log likelihood and its derivatives are dominated by a square--integrable envelope function 
%     %       for all $\theta$ in a neighborhood of $\theta_\infty$.
%     % \end{itemize}
% \end{itemize}
%

\theorem{
    \textbf{Theorem 3 \citep{giordano:2023:bayesij} (sketch): }\\
    Under Bernstein-von Mises-like conditions,
    % \textbf{(Consistency of the von--Mises expansion in finite dimensions) }
    %
\begin{align*}
    \sup_{\ttil \in [0,1]} | \red{\err(\ttil)} | \overset{p}{\rightarrow} 0
    \quad\textrm{in the Bayesian von--Mises expansion}.
\end{align*}
%
}

\theorem{
    \textbf{Corollary: }Let 
         $\meansur (\infl_i - \overline{\infl})^2 \overset{p}{\rightarrow} V$.
By the CLT applied to $\infl_i$,
\begin{align*}
     \sqrt{\nsur} \left(\expect{\post}{\phi(\theta)} - \T(\psur, \nsur) \right)
     \overset{d}{\rightarrow}
     \mathcal{N}\left(
        0, V
     \right).
\end{align*}
}

\pause

To compute the parametric bootstrap instead is a simple change:
%
$$
\posteps \propto 
\post \exp\left(\sumsur \epsilon 
    \left( \ell(\y^*_i \vert \x_i, \theta) - 
           \ell(\y_i \vert \x_i, \theta)\right) \right).
$$
% 
(See, e.g., \citet{giordano:2026:mrplew} Theorem 4.1).


\end{frame}
