\begin{frame}[t]{Why flows for sensitivity analysis?}
%
Current Bayesian local robustness is built around linearizing posterior expectations:
$$
\expect{\p^\epsilon(\theta)}{\phi(\theta)} \approx
\expect{\p(\theta)}{\phi(\theta)} +
\fracat{\partial \expect{\p^\epsilon(\theta)}{\phi(\theta)}}
       {\partial \epsilon}{\epsilon = 0}(\epsilon - 0),
$$
%
where the derivative is estimated --- with noise --- from MCMC draws
\citep{gustafson:1996:localposterior,giordano:2018:covariances,cabral:2025:RobustnessModelChecking2025,dinoia:2025:LikelihoodDistortionBayesian2025,giordano:2026:mrplew}.

This has some shortcomings:
%
\begin{itemize}
\item Potential MCMC variance, not obvious how to smooth
\item Linear extrapolation can lead to impossible values (e.g.~negative variances)
\item Complex quantities like posterior quantiles are hard to linearize
\item Bayesians like to work with draws, not analytic expectations (CITE)
\end{itemize}
%

\textbf{Idea:} Instead of linearizing expectations, learn a flow
from $\p$ to $\p^\epsilon$.\\[1em]

There's a large literature on learning flows between distant
distributions.  (CITE)\\
Our problem is easier:\\
%
\begin{itemize}
\item Only need ``local'' flows
\item Our perturbations have a lot of structure (e.g.~bootstraps)
\end{itemize}
%


\end{frame}

\begin{frame}[t]{Flow}

Consider $\peps(\theta) \propto \p(\theta) \exp(\epsilon \h(\theta))$.

We want a flow $\theta \mapsto \feps(\theta)$ such that,
for small $\epsilon$, $\feps(\theta) \sim \peps(\theta)$
and $\f^0(\theta) = \theta$.\\[1em]

\pause

Suppose there is a scalar-valued potential function $\v(\theta)$ such that
$$
\feps(\theta) = \theta + \epsilon \nabla \v(\theta) + O(\epsilon^2)
\quad\textrm{as }\epsilon \rightarrow 0. \textrm{ Then:}
$$
%
\pause
$$
\begin{aligned}
  \expect{\peps(\theta)}{\phi(\theta)} ={}& \expect{\p(\theta)}{\phi(\feps(\theta))}
\Rightarrow & \textrm{(good flow)} \pause \\
\fracat{\partial}{\partial \epsilon}{\epsilon = 0}  
  \expect{\peps(\theta)}{\phi(\theta)} ={}&
\fracat{\partial}{\partial \epsilon}{\epsilon = 0}
  \expect{\p(\theta)}{\phi(\theta + \epsilon \nabla \v(\theta))} \Rightarrow
  & \textrm{(smoothness)} \pause 
\\
% \\
% \verteq \quad\quad\quad & \quad\quad\quad \verteq
% \\
%\cov{\p(\theta)}{\phi(\theta), \h(\theta)}
(\star) \quad\quad\quad 
\cov{\p(\theta)}{\phi(\theta), \h(\theta)}
={}&
  \expect{\p(\theta)}{\nabla \phi(\theta)^\trans 
                      \nabla \v(\theta) }
  & \textrm{(Connecting $\h$ and $\v$)}
\\
\end{aligned}
$$
%
\pause
%
\theorem{\textbf{Theorem (Schewengber 2026): } \\
% Assume that
% %
% \begin{itemize}
% \item $\log \p(\theta) \in C^2(\mathcal{R}^d)$
% \item  $\lim_{\theta \to \infty} \left( \frac{1}{2}\| \nabla \log
%        \p(\theta) \|^2 - \Delta \log \p(\theta) \right) = +\infty$.
% \item $\expect{\p(\theta)}{\h(\theta)^2} < \infty$
% \end{itemize}
%
Under mild regularity conditions there exists a solution $\v$ to the differential
equation
$$
\lotto(\v) :={}
- \nabla \v^\trans \nabla \log \p 
    - \Delta \v = \h
$$
such that  $\norm{\nabla \v}_{L^2(\p)} <  \infty$ and
the flow $\theta_\epsilon = \theta + \epsilon \nabla \v(\theta)$
satsifes Equation $(\star)$.
%
}

Note that $\lotto$ is well-known: it's the drift operator, generator of Langevin
diffusion, and the trace of the Stein
operator.\citep{ambrosio2021lectures,song2021scorebasedgenerativemodelingstochastic,ross2011fundamentals}


\end{frame}

\begin{frame}[t]{How can we find $\v(\theta)$ in practice?}

Solving PDEs is hard.  How can we easily find $\v(\theta)$ in practice?
$$
\begin{aligned}
(\star) \quad\quad\quad 
\cov{\p(\theta)}{\phi(\theta), \h(\theta)}
={}&
  \expect{\p(\theta)}{\nabla \phi(\theta)^\trans 
                      \nabla \v(\theta) }
  & \textrm{(Identity connecting $\h$ and $\v$)}
\end{aligned}
$$
Basis expansion approach (Otto's calculus \citep{otto_geometry_2001}):
%
\begin{itemize}
\item Specify a family of real-valued test functions $\phi_n(\theta)$ for $n \in [N]$
\item Specify a family of real-valued basis functions $\b_m(\theta)$ for $m \in [M]$
\end{itemize}
%
Let $\v(\theta; \beta) := \sum_{m=1}^M \beta_m \b_m(\theta)$ and
plug into Equation $(\star)$.

\pause
%
Stacking in matrices:
$$
\begin{aligned}
\underbrace{
\begin{bmatrix}
  \vdots\\
  \cov{\p(\theta)}{\phi_n(\theta), \h(\theta)} \\
  \vdots\\
\end{bmatrix}%_{n \in [N]}
}_{
  \shortstack{$N \times 1$ matrix}
}
%
%
\underbrace{
\begin{bmatrix}
  & \vdots & \\
\hdots &
\expect{\p(\theta)}{\nabla \phi_n(\theta)^\trans 
                      \nabla \b_m(\theta) } &
\hdots \\
  & \vdots & \\
\end{bmatrix}%_{n \in [N], m \in [M]}
}_{
  \shortstack{$N \times M$ matrix}
}
\begin{bmatrix}
  \vdots \\
  \beta_m \\
  \vdots
\end{bmatrix}%_{m \in [M]}.
\end{aligned}
$$
%

\begin{frame}[t]{An elegant flow}

Suppose we believe that $\post$ is approximately Gaussian.

Take the basis functions $\b_m(\theta) = \theta_m$ (1st Hermite polynomial)
and test functions $\phi_n(\theta) = \b_m(\theta)$.  Then the optimal
flow is given by
$$
\theta_\epsilon = \theta +
  \epsilon \underbrace{\cov{\post}{\theta, \h(\theta)}}_{
    \fracat{\partial \expect{\post}{\theta}}{\partial \epsilon}{\epsilon = 0}
  }
$$
%
\textbf{This just translates the posterior in the direction of the mean changes.}
%
\pause

Suppose we have $\theta_m \sim \post$, and bootstrap
weights $\w_i$.  Set
%
$$
\theta_m^* = \theta_m + \meansur \w_i \infl_i
  % \underbrace{\covhat{\post}{\theta, \ell(\y_i \vert \x_i, \theta)}}_{
  %   \textrm{Influence score }\infl_i / N
  %   }
$$
%
and compute any posterior quantities from these samples.
\end{frame}
