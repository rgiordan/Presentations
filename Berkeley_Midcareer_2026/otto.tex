\begin{frame}[t]{Why flows for sensitivity analysis?}
%
The IJ is built around linearizing posterior expectations:
$$
\expect{\p^\epsilon(\theta)}{\g(\theta)} \approx
\expect{\p}{\g(\theta)} +
\fracat{\partial \expect{\p^\epsilon(\theta)}{\g(\theta)}}
       {\partial \epsilon}{\epsilon = 0}(\epsilon - 0),
$$
%
where the derivative is estimated --- with noise --- from MCMC draws.

This has some shortcomings:
%
\begin{itemize}
\item Potential MCMC variance, not obvious how to smooth
\item Linear extrapolation can lead to impossible values (e.g.~negative variances)
\item Complex quantities like posterior quantities are hard to linearize
\item Bayesians like to work with draws, not analytic expectations
\end{itemize}
%

\textbf{Idea:} Instead of linearizing expectations, learn a flow
from $\p$ to $\p^\epsilon$.\\[1em]

There's a large literatue on learning flows between distant
distributions.  Our problem is easier:\\
%
\begin{itemize}
\item Only need ``local'' flows
\item Our perturbations have a lot of structure (e.g.~bootstraps)
\end{itemize}
%


\end{frame}

\begin{frame}[t]{Flow}

\sensitivitysetup{}

We want a flow $\theta \mapsto \feps(\theta)$ such that,
for small $\epsilon$, $\feps(\theta) \sim \peps(\theta)$
and $\f^0(\theta) = \theta$.\\[1em]

\hrulefill

Suppose there is a scalar-valued potential function $\v(\theta)$ such that
$$
\feps(\theta) = \theta + \epsilon \nabla \v(\theta) + O(\epsilon^2)
\quad\textrm{as }\epsilon \rightarrow 0. \textrm{ Then:}
$$
%
$$
\begin{aligned}
  \expect{\peps(\theta)}{\phi(\theta)} ={}& \expect{\p(\theta)}{\phi(\feps(\theta))}
\Rightarrow \\
\fracat{\partial}{\partial \epsilon}{\epsilon = 0}  
  \expect{\peps(\theta)}{\phi(\theta)} ={}&
\fracat{\partial}{\partial \epsilon}{\epsilon = 0}
  \expect{\p(\theta)}{\phi(\theta + \epsilon \nabla \v(\theta))}
\\
\verteq \quad\quad\quad & \quad\quad\quad \verteq
\\
%\cov{\p(\theta)}{\phi(\theta), \h(\theta)}
\expect{\p(\theta)}{\phi(\theta) \h(\theta)}
={}&
  \expect{\p(\theta)}{\nabla \phi(\theta)^\trans 
                      \nabla \v(\theta) }\\
\end{aligned}
$$

\hrulefill

%
\begin{itemize}
\item When does such a $\v(\theta)$ exist?
\item How can we find it in practice?
\end{itemize}
%

\end{frame}

\def\lotto{\mathcal{L}_{\p}}

\begin{frame}[t]{When does such a $\v(\theta)$ exist?}

\sensitivitysetup{}

We want a flow $\theta \mapsto \feps(\theta)$ such that,
for small $\epsilon$, $\feps(\theta) \sim \peps(\theta)$
and $\f^0(\theta) = \theta$.\\[1em]

\textbf{When does such a $\v(\theta)$ exist?}

\theorem{\textbf{Theorem (Schewengber 2026): } Assume that
%
\begin{itemize}
\item $\log \p(\theta) \in C^2(\mathcal{R}^d)$
\item  $\lim_{\theta \to \infty} \left( \frac{1}{2}\| \nabla \log
       \p(\theta) \|^2 - \Delta \log \p(\theta) \right) = +\infty$.
\item $\expect{\p(\theta)}{\h(\theta)^2} < \infty$
\end{itemize}
%
Then there exists a solution $\v$ to the differential
equation
$$
-\lotto(\v) :={}
\nabla \v^\trans \nabla \log \p 
    + \Delta \v = \h
$$
such that  $\norm{\nabla \v}_{L^2(\p)} <  \infty$, and
the flow $\theta_\epsilon = \theta + \epsilon \nabla \v(\theta)$
satsifes $\theta_\epsilon \sim \peps(\theta)$ for small $\epsilon$.
%
}

Note that $\lotto$ is the drift operator, generator of Langevin diffusion,
trace of the Stein operator.

\end{frame}

\begin{frame}[t]{How can we find $\v(\theta)$ in practice?}

Fix $\h(\theta)$.  Specify
%
\begin{itemize}
\item A family of real-valued test functions $\phi_n(\theta)$ for $n \in [N]$
\item A family of real-valued basis functions $\b_m(\theta)$ for $m \in [M]$
\end{itemize}
%
$$
\textrm{Let }\v(\theta; \beta) := \sum_{m=1}^M \beta_m \b_m(\theta).
\textrm{ Then}
$$
%
$$
\begin{aligned}
\underbrace{
  \expect{\p(\theta)}{\phi_n(\theta) \h(\theta)}
}_{\textrm{Estimate with MCMC!}}
={}&
\sum_{m=1}^M
  \beta_m
\underbrace{
  \expect{\p(\theta)}{\nabla \phi_n(\theta)^\trans 
                      \nabla \b_m(\theta) }
}_{\textrm{Estimate with MCMC!}}
\end{aligned}
$$
This gives $N$ linear equations in $M$ unknowns.
Known as ``Otto's calculus'' in differential equations.   Formally
similar to numerical PDE solutions known as Galerkin methods,
except we have a stochastic grid ($\theta \sim \p(\theta)$) 
rather than a deterministic one.
% $$
% F = B \beta
% $$

Unlike standard linear regression:\\[1em]
%
\begin{itemize}
\item We are free to choose $\phi_n$ and $\b_m$
\item Both $F$ and $B$ are measured with noise
\item We have a lot of information about the covariance of $F$ and $B$
\end{itemize}
%
%
\theorem{ \textbf{Theorem (Schewengber 2026, sketch): } The solution
gives the best L2 approximation to $\norm{\nabla \v(\theta)}_{L^2(\p)}$
in the span of $\phi_n(\theta)$.  When $\p$ is Gaussian, the ordered Hermite
polynomials give the best finite-rank approximation.
}

\end{frame}

\begin{frame}[t]{An elegant flow}

Suppose we believe that $\post$ is approximately Gaussian.

Take the basis functions $\b_m(\theta) = \theta_m$ (1st Hermite polynomial)
and test functions $\phi_n(\theta) = \b_m(\theta)$.  Then the optimal
flow is given by
$$
\theta_\epsilon = \theta +
  \epsilon \underbrace{\cov{\post}{\theta, \h(\theta)}}_{
    \fracat{\partial \expect{\post}{\theta}}{\partial \epsilon}{\epsilon = 0}
  }
$$
%
\textbf{This just translates the posterior in the direction of the mean changes.}
%

Suppose we have $\theta_m \sim \post$, and bootstrap
weights $\w_i$.  Set
%
$$
\theta_m^* = \theta_m + \meansur \w_i \infl_i
  % \underbrace{\covhat{\post}{\theta, \ell(\y_i \vert \x_i, \theta)}}_{
  %   \textrm{Influence score }\infl_i / N
  %   }
$$
%
and compute any posterior quantities from these samples.
\end{frame}
