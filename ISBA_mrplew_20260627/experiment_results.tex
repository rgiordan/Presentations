\begin{frame}[t]{Same-Sex Marriage subgroup contribution}

Note that both methods use \emph{national data} to infer public opinion \emph{in California}.

\begin{block}{Question:}
    How do responses in Texas affect our estimate of public opinion in CA?
\end{block}

\begin{block}{More specific question:}
    Suppose we \emph{increased} the responses $\y_i$ in Texas.

    How would our estimate of public opinion in CA change?
\end{block}

\MapQuery{}



\begin{block}{Theorem: \parencite{giordano:2026:mrplew} (informal)}
    Take $\surcol{\ytil_i = \y_i + \delta \f(\x_i)}$.
    We state conditions under which
$$
\onslide<4->{\sup_{f \in \mathcal{F}}}
\abs{
    \muhatmrp[\Ytil] - \muhatmrp
    - \delta \surcol{\sumsur \w_i^\mrp \f(\x_i)}
}
\only<1>{ = \textrm{Small}}
\only<2->{ = O(\delta^2)}
\onslide<3->{\textrm{ as }N \rightarrow \infty}
$$
\onslide<4->{
    ...for a very broad class of $\mathcal{F}$.
    \footnote{$\mathcal{F}$ can be any Donsker class of measurable functions
    with uniformly bounded
    $\expect{}{\norm{\x} \f(\x)}_2^2 $.}
}
\end{block}

\textbf{Uniformity helps justify searching over a large class of perturbations $\f$.}





\end{frame}
