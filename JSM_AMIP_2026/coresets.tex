\begin{frame}[t]{Do we identify ``the most important datapoints?''}

\begin{exampleblock}{}
``Unraveling the intricate connections between data and model predictions
is critical in machine learning...''
  \hspace*\fill{\small \textcite{hu:2024:MostInfluentialSubset}}
\end{exampleblock}

\onslide<2->{
Let $\S$ denote sets of data, $[N]$ denote all points, and $\delta(\S)$ the
change induced when dropping $\S$.\\[1em]

Informally, we use the influence function (IF) to solve:
$$
\S_{AMIP} := \textrm{The smallest $\S$ with $\delta(\S) \ge \Delta$}
$$
}

\onslide<3->{
Contrast this with the problem of finding a
``coreset:''\\ 
\parencite{
    feldman:2011:UnifiedFrameworkApproximating2011,
    chen:2012:SuperSamplesKernelHerding2012,
    ting:2018:OptimalSubsamplingInfluence2018,
    chen:2024:CoresetMarkovChain2024}
$$
\begin{aligned}
\S_{CORE} :={}& \textrm{The smallest $\S$ with $\delta([N] \setminus \S) \le \Delta$}.
\end{aligned}
$$
%
%\vspace{1em}
%
\onslide<4->{
Both $\S_{AMIP}$ and $\S_{CORE}$ are ``important points,'' but in different senses. 
}
}
\end{frame}

\begin{frame}[t]{Do we identify ``the most important datapoints?''}

\textbf{Bad idea:} Use the IF to drop a large \% of low-influence points.\\[1em]

\onslide<2->{
\centering
\begin{minipage}{0.85\textwidth}
\GreedyDrop{}
\end{minipage}

\textbf{Problem:} The IF is not accurate for large \% of dropped data.
}

\end{frame}

\begin{frame}[t]{Do we identify ``the most important datapoints?''}

\textbf{Better idea:} Use the IF to reach a coreset with ``gradient descent''.\\[1em]

\onslide<2->{
\centering
\begin{minipage}{0.85\textwidth}
\GDPath{}
% \only<3>{\GDRefit{}}
\end{minipage}

\onslide<1->{
\textbf{Problem:} You have to re-fit at each step, and eventually
run out of low-influence points.
}

(See also \textcite{ting:2018:OptimalSubsamplingInfluence2018} for another way to use
the IF for coreset construction.)
}

\end{frame}
